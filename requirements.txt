# FlameAnalyzer inference backend - runtime dependencies.
# Only packages that are actually imported by the backend are listed.

# --- HTTP API (app/api.py) ---------------------------------------------
# FastAPI app served by `uvicorn app.api:app`; the Starlette TestClient used
# by the API tests ships with FastAPI.
fastapi>=0.110.0

# ASGI server. Standard (non-`[standard]`-extra) build keeps the dependency
# list small; use `uvicorn[standard]` in production for uvloop/httptools.
uvicorn>=0.27.0

# Required by FastAPI to parse `multipart/form-data` uploads.
python-multipart>=0.0.9

# --- Inference core ----------------------------------------------------
# YOLO inference (pulls in torch, torchvision, numpy, opencv, ...)
ultralytics>=8.3.0

# Deep-learning runtime. Listed explicitly so the CUDA build is pinned by the
# deployment image rather than resolved implicitly.
# For Modal, install the CUDA wheel that matches the container image instead.
torch>=2.2.0

# Headless OpenCV: no GUI/X11 libraries, safe in containers.
opencv-python-headless>=4.8.0

# Array maths and LAB conversion (skimage.color.rgb2lab / lab2rgb)
numpy>=1.24.0
scikit-image>=0.22.0

# K-Means and Gaussian Mixture clustering
scikit-learn>=1.3.0
